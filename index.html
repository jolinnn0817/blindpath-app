function speak(text) { if (!("speechSynthesis" in window)) { showMessage( "navigationMessage", "Voice guidance is not supported by this browser." ); return; } window.speechSynthesis.cancel(); var utterance = new SpeechSynthesisUtterance(text); utterance.lang = byId("languageSelect").value; utterance.rate = Number(byId("speechRate").value) || 1; utterance.onerror = function () { showMessage( "navigationMessage", "Voice guidance could not play. Please press Read guidance to try again." ); }; window.speechSynthesis.speak(utterance); }
